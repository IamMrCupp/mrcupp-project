{{- /* Open Graph + Twitter card tags, so shared links render as cards on Facebook, Threads, Discord, Slack, etc. Image: front-matter `image` (a path under static/) wins; otherwise the site logo. The logo is square, so it gets a "summary" card; a post image gets the large one. Kept as its own partial so the planned hugo-plugins module can replace it wholesale. */ -}} {{- $desc := "" -}} {{- with .Description }}{{ $desc = . }}{{ else }}{{ with .Params.subtitle }}{{ $desc = . }}{{ else }}{{ if and .IsPage .Summary }}{{ $desc = .Summary | plainify | htmlUnescape | truncate 200 }}{{ else }}{{ with .Site.Params.subtitle }}{{ $desc = . }}{{ end }}{{ end }}{{ end }}{{ end -}} {{- $title := cond .IsHome .Site.Title .Title -}} {{- $img := "" -}}{{- $large := false -}} {{- with .Params.image }}{{ $img = . | absURL }}{{ $large = true }}{{ else }}{{ with .Site.Params.logo }}{{ $img = . | absURL }}{{ end }}{{ end -}} {{ with $desc }}{{ end }} {{- if and .IsPage (eq .Type "post") }} {{ if not .Lastmod.IsZero }}{{ end }} {{ with .Site.Params.Author.name }}{{ end }} {{ range .Params.tags }} {{ end }} {{- else }} {{- end }} {{ with $img }} {{ end }} {{ with $desc }}{{ end }}
